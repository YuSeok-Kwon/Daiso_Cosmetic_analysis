// All ID and count fields are non-negative integers.
// VARCHAR(n) lengths reflect expected maximum values with buffer.

// =======================
// Reviews domain
// =======================
Table Reviews_core {
  review_id int [primary key]
  user_id int [not null]
  product_code int [not null]
  promotion_id int
  rating smallint [not null]
  image_count smallint [not null]
  review_date date [not null]
  is_reorder boolean [not null]
}

Table Reviews_text {
  review_id int [primary key] // PK/FK (1:1 with Reviews_core)
  text text
  review_length smallint [not null]
}

// =======================
// Users domain
// =======================
Table User_id_map {
  user_masked varchar(30) [primary key] // 마스킹된 유저 닉네임
  user_id int [not null, unique]
}

Table Users_profile {
  user_id int [primary key]
  user_total_reviews int [not null]
  user_activity_level varchar(20) [not null]
  user_rating_tendency varchar(30) [not null]
  review_tenure int [not null]
}

Table Users_repurchase {
  user_id int [not null]
  reorder_user_category int [not null]
  reorder_user_brand int [not null]
  reorder_user_avg_rating float [not null]
}

// =======================
// Products domain
// =======================
Table Products_core {
  product_code int [primary key]
  manufacturer_id smallint [not null]
  brand_id smallint [not null]
  name varchar(80) [not null]
  price int [not null]
  country varchar(20) [not null]
}

Table Products_stats {
  product_code int [primary key] // PK/FK (1:1 with Products_core)
  likes int [not null]
  shares int [not null]
  review_count int [not null]
  first_review_date date // 첫 리뷰 작성일 (리뷰 없으면 NULL)
  engagement_score float [not null]
  cp_index float [not null]
  review_density float [not null]
}

Table Products_category {
  product_code int [primary key] // PK/FK (1:1 with Products_core)
  category_1 varchar(20) [not null]
  category_2 varchar(30) [not null]
}

Table Functional {
  product_code int [not null]
  ITEM_pH float
  pH_category varchar(15)
  is_whitening boolean [not null]
  is_wrinkle_reduction boolean [not null]
  is_sunscreen boolean [not null]
  is_acne boolean [not null]
}

Table Brand {
  brand_id smallint [primary key]
  brand_name varchar(30) [not null]
}

Table Manufacturer {
  manufacturer_id smallint [primary key]
  ENTP_NAME varchar(40) [not null]
}

// =======================
// Ingredients domain
// =======================
Table Ingredients_dic {
  ingredient_id smallint [primary key]
  ingredient_name varchar(60) [not null]
  ingredient_type varchar(30) [not null]
  effect varchar(30) [not null]
  is_allergic boolean [not null]
}

Table products_ingredients {
  product_code int [not null]
  ingredient_id smallint [not null]

  Indexes {
    (product_code, ingredient_id) [unique]
  }
}

// =======================
// (Optional) Promotions domain (kept, but you can delete if not needed)
// =======================
Table Promotions {
  promotion_id int [primary key]
  start_date date [not null]
  end_date date [not null]
  description varchar(100) [not null]
  brand_id smallint
  event_type varchar(30) [not null]
}

// =======================
// References (relationships)
// =======================

// Reviews -> Products / Users
Ref reviews_product: Reviews_core.product_code > Products_core.product_code // many-to-one
Ref reviews_user: Reviews_core.user_id > Users_profile.user_id

// Reviews_text (1:1)
Ref review_text: Reviews_text.review_id - Reviews_core.review_id // one-to-one

// User_id_map -> Users_profile (1:1)
Ref user_map: User_id_map.user_id - Users_profile.user_id // one-to-one

// Users_repurchase (1:1)
Ref User_repurchase: Users_repurchase.user_id - Users_profile.user_id

// Products_core -> Brand / Manufacturer
Ref product_brand: Products_core.brand_id > Brand.brand_id                 // many-to-one
Ref product_manufacturer: Products_core.manufacturer_id > Manufacturer.manufacturer_id // many-to-one

// Product 1:1 extension tables
Ref product_stats: Products_stats.product_code - Products_core.product_code       // one-to-one
Ref product_category: Products_category.product_code - Products_core.product_code // one-to-one
Ref product_functional: Functional.product_code > Products_core.product_code      // one-to-one

// Ingredients bridge (many-to-many)
Ref pi_product: products_ingredients.product_code > Products_core.product_code    // many-to-one
Ref pi_ingredient: products_ingredients.ingredient_id > Ingredients_dic.ingredient_id // many-to-one

// Brand (1) -> Promotions (0..N), Promotions.brand_id can be NULL
Ref brand_promotions: Promotions.brand_id > Brand.brand_id // many-to-one (nullable FK on Promotions)

// Promotions (1) -> Reviews_core (0..N), Reviews_core.promotion_id can be NULL
Ref promotion_reviews: Reviews_core.promotion_id > Promotions.promotion_id // many-to-one (nullable FK on Reviews_core)
